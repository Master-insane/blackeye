import time
import queue
import networkx as nx
import matplotlib.pyplot as plt

# BFS with Goal Support
def bfs(G, start_node, goal_node=None):
    visited = set()
    q = queue.Queue()
    q.put(start_node)
    order = []

    while not q.empty():
        vertex = q.get()
        if vertex not in visited:
            visited.add(vertex)
            order.append(vertex)

            # Stop if goal is reached
            if goal_node is not None and vertex == goal_node:
                return order

            for node in G[vertex]:
                if node not in visited:
                    q.put(node)
    return order

# Visualization Function
def visualise_graph(G, pos, title, order, start_node, goal_node):
    for current in order:
        plt.clf()  # clear frame
        plt.title(f"{title}\nVisiting: {current}")

        node_colors = []
        for node in G.nodes():
            if node == current:
                node_colors.append('red')  # current exploring node
            elif node == start_node:
                node_colors.append('blue')  # start
            elif node == goal_node:
                node_colors.append('yellow')  # goal
            else:
                node_colors.append('green')  # others

        nx.draw(G, pos, with_labels=True, node_size=1800, node_color=node_colors, font_size=12)
        plt.draw()
        plt.pause(1.5)  # delay to show steps

    plt.show()

# Create Graph
G = nx.Graph()
G.add_edges_from([
    ('A','B'), ('A','C'),
    ('B','D'), ('B','E'),
    ('C','F'), ('C','G')
])

# Position for drawing
pos = nx.spring_layout(G)

# Run BFS from A to F
order = bfs(G, start_node='A', goal_node='F')

# Visualize
visualise_graph(G, pos, title='BFS Traversal (A to F)', order=order, start_node='A', goal_node='F')
